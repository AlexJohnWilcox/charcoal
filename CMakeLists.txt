cmake_minimum_required(VERSION 3.16)
project(charcoal CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra)

file(GLOB CORE_SRC CONFIGURE_DEPENDS src/*.cc)
if(CORE_SRC)
  add_library(charcoal_core STATIC ${CORE_SRC})
  target_include_directories(charcoal_core PUBLIC include src)
endif()

if(EXISTS ${CMAKE_SOURCE_DIR}/cli/main.cc AND TARGET charcoal_core)
  add_executable(charcoal cli/main.cc)
  target_link_libraries(charcoal PRIVATE charcoal_core)
endif()

enable_testing()
file(GLOB TEST_SRC CONFIGURE_DEPENDS tests/*.cc)
if(TEST_SRC AND TARGET charcoal_core)
  add_executable(charcoal_tests ${TEST_SRC})
  target_link_libraries(charcoal_tests PRIVATE charcoal_core)
  target_include_directories(charcoal_tests PRIVATE tests)
  add_test(NAME charcoal_tests COMMAND charcoal_tests)
endif()

# Fuzzers only when a fuzzing engine is provided (ClusterFuzzLite / local).
if(DEFINED ENV{LIB_FUZZING_ENGINE} OR CHARCOAL_FUZZ)
  foreach(F module_fuzzer compile_fuzzer)
    if(EXISTS ${CMAKE_SOURCE_DIR}/fuzz/${F}.cc AND TARGET charcoal_core)
      add_executable(${F} fuzz/${F}.cc)
      target_link_libraries(${F} PRIVATE charcoal_core)
      target_compile_options(${F} PRIVATE -fsanitize=fuzzer-no-link)
      target_link_options(${F} PRIVATE -fsanitize=fuzzer)
    endif()
  endforeach()
endif()
